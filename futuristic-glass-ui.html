// Angular 17/18/19 Standalone Component: fg-toggle.component.ts
import { Component, input, model, output } from '@angular/core';

@Component({
  selector: 'fg-toggle',
  standalone: true,
  template: `
    <div class="fg-toggle-row">
      <div class="fg-toggle-label">
        <span class="fg-toggle-title">{{ title() }}</span>
        @if (subtitle()) {
          <span class="fg-toggle-subtitle">{{ subtitle() }}</span>
        }
      </div>
      <label class="fg-toggle">
        <input type="checkbox" [checked]="checked()" (change)="onToggle($event)" />
        <span class="fg-toggle__track">
          <span class="fg-toggle__thumb"></span>
        </span>
      </label>
    </div>
  `
})
export class FgToggleComponent {
  title = input.required<string>();
  subtitle = input<string>();
  checked = model<boolean>(false);

  onToggle(event: Event) {
    const isChecked = (event.target as HTMLInputElement).checked;
    this.checked.set(isChecked);
  }
}
// Angular Standalone Component: fg-segmented.component.ts
import { Component, input, model, viewChild, ElementRef } from '@angular/core';

@Component({
  selector: 'fg-segmented',
  standalone: true,
  template: `
    <div class="fg-segmented" #container>
      <div class="fg-segmented__indicator" [style.transform]="indicatorTransform()" [style.width.px]="indicatorWidth()"></div>
      @for (opt of options(); track opt; let idx = $index) {
        <div class="fg-segmented__item" [class.fg-segmented__item--active]="selected() === idx" (click)="select(idx, $event)">
          {{ opt }}
        </div>
      }
    </div>
  `
})
export class FgSegmentedComponent {
  options = input.required<string[]>();
  selected = model<number>(0);
  indicatorTransform = signal<string>('translateX(0)');
  indicatorWidth = signal<number>(0);

  select(index: number, event: MouseEvent) {
    this.selected.set(index);
    const target = event.currentTarget as HTMLElement;
    this.indicatorWidth.set(target.offsetWidth);
    this.indicatorTransform.set(`translateX(${target.offsetLeft - 4}px)`);
  }
}
// Angular Standalone Component: fg-dropdown.component.ts
import { Component, input, model, signal } from '@angular/core';

@Component({
  selector: 'fg-dropdown',
  standalone: true,
  template: `
    <div class="fg-dropdown" [class.fg-dropdown--open]="isOpen()">
      <button class="fg-dropdown__trigger" type="button" (click)="toggle()">
        <span>{{ selectedLabel() }}</span>
        <span class="fg-dropdown__chevron">▼</span>
      </button>
      <div class="fg-dropdown__menu">
        @for (item of items(); track item.value) {
          <button class="fg-dropdown__item" [class.fg-dropdown__item--selected]="item.value === value()" (click)="select(item)">
            <span>{{ item.label }}</span>
            @if (item.value === value()) { <span>✓</span> }
          </button>
        }
      </div>
    </div>
  `
})
export class FgDropdownComponent {
  items = input.required<{ label: string; value: any }[]>();
  value = model<any>();
  isOpen = signal(false);

  toggle() { this.isOpen.update(v => !v); }
  select(item: { label: string; value: any }) {
    this.value.set(item.value);
    this.isOpen.set(false);
  }
}
